Cloudinary Product Gallery

A full-featured product gallery widget — images, video, and 360° spin — embedded in an Angular app using the Cloudinary Gallery Widget .

How it works

The gallery widget is initialised inside ngAfterViewInit and torn down in ngOnDestroy, targeting a native DOM element via ViewChild. 

// product-gallery.component.ts
ngAfterViewInit(): void {
  this.widget = cloudinary.galleryWidget({
    container: this.galleryContainer.nativeElement,
    cloudName: 'demo',
    mediaAssets: [
      { tag: 'electric_car_product_gallery_demo' },
      { tag: 'electric_car_product_gallery_demo', mediaType: 'video' },
      { tag: 'electric_car_360_product_gallery_demo', mediaType: 'spin' },
    ],
  });
  this.widget.render();
}

ngOnDestroy(): void {
  this.widget?.destroy();
}

Add the widget script to index.html:

<script src="https://product-gallery-v2.cloudinary.com/2.0.1/all.js"></script>
